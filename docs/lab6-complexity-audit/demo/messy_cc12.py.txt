"""Demo fixture for the pre-commit FAIL scenario: deeply nested conditions, CC = 12."""


def shipping_label(weight: float, express: bool, country: str, fragile: bool, insured: bool) -> str:
    if weight > 0:
        if express:
            if country == "UA":
                if fragile:
                    return "UA-EXPRESS-FRAGILE"
                if insured:
                    return "UA-EXPRESS-INSURED"
                return "UA-EXPRESS"
            if country == "PL":
                if fragile:
                    return "PL-EXPRESS-FRAGILE"
                if insured:
                    return "PL-EXPRESS-INSURED"
                return "PL-EXPRESS"
            return "INTL-EXPRESS"
        if country == "UA":
            if fragile:
                return "UA-FRAGILE"
            if insured:
                return "UA-INSURED"
            return "UA-STANDARD"
        return "INTL-STANDARD"
    return "INVALID"
