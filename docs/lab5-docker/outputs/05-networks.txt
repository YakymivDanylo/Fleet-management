# backend is internal: db/redis are not published to the host
$ docker network inspect fleet_lab5_backend --format "internal={{.Internal}}"
internal=true containers=fleet_lab5-redis-1 fleet_lab5-worker-1 fleet_lab5-rabbitmq-1 fleet_lab5-db-1 fleet_lab5-api-1 
$ docker network inspect fleet_lab5_edge
internal=false containers=fleet_lab5-rabbitmq-1 fleet_lab5-api-1 

# api resolves db by service name over the backend network
$ docker compose exec api python -c "socket.create_connection((db,5432))"
api -> db:5432 OK

# worker is backend-only: no route to the internet
$ docker compose exec worker python -c "socket.create_connection((1.1.1.1,80))"
worker -> internet BLOCKED: [Errno 101] Network is unreachable

# db port is not published on the host
$ python -c "socket.create_connection((localhost,5432))"
host -> db:5432 refused: timed out
